here is the first example card I made: 

// Creating the base card
    //   2      3     4
    //    -------------
    //   |      |      |
    //   |      |      |
    //   |      |      |
    //   |      |      | 
    //    -------------
    //   1      0      5
    //        origin      (for now!!)
    //          cf        (cf = central fold)
    0
    std::vector<Eigen::Vector3f> baseCardPoints = {
        Eigen::Vector3f{0.0, -1.0, 0.0},  //0
        Eigen::Vector3f{-1.0, -1.0, 0.0}, //1
        Eigen::Vector3f{-1.0, 1.0, 0.0}, //2
        Eigen::Vector3f{0.0, 1.0, 0.0},  //3
        Eigen::Vector3f{1.0, 1.0, 0.0},  //4
        Eigen::Vector3f{1.0, -1.0, 0.0}   //5
    };

    std::vector<Crease> CentralFold = {
        Crease {0, 3, CreaseType::Valley}
    };

    std::vector<GlueDot> emptyGlues;

    Mechanism baseCard(baseCardPoints, CentralFold, emptyGlues, true, 0);

    // Mechanism 1, a simple right angle v-fold
    //
    //    ____s____        2____3____4
    //   |    |    |       |    |    |
    //   |____|____|       |____|____|
    //   g    g    g       1    0    5
    //bll  bl/br  brr    (bl = bottomLeft, b = bottom, br = bottomRight)
    //      o            (o = origin)           
    //      ^ crease line


    std::vector<Eigen::Vector3f> mech1Points = {
        Eigen::Vector3f{5.0, 0.0, 0.0},  //0
        Eigen::Vector3f{4.0, 0.0, 0.0},  //1
        Eigen::Vector3f{4.0, 1.0, 0.0},  //2
        Eigen::Vector3f{5.0, 1.0, 0.0},  //3
        Eigen::Vector3f{6.0, 1.0, 0.0},  //4
        Eigen::Vector3f{6.0, 0.0, 0.0},  //5
    };

    std::vector<Crease> mech1Crease = {
        // Crease {i, j, type}
        Crease {0, 3, CreaseType::Mountain}
    };

    float d = std::sqrt(2.0)/2.0;
    std::vector<GlueDot> mech1Glues = { // this part is tricky because it's up to the artists to make sure that lengths stay the same 
        // GlueDot {i, n, gluePosition, weights}
        GlueDot {1, 0, Eigen::Vector3f{-d, d, 0.0}, Eigen::Vector3f{0.0,0.0,0.0}},
        GlueDot {0, 0, Eigen::Vector3f{0.0, 0.0, 0.0}, Eigen::Vector3f{0.0,0.0,0.0}},
        GlueDot {5, 0, Eigen::Vector3f{d, d, 0.0}, Eigen::Vector3f{0.0,0.0,0.0}},
    };

    Mechanism mech1(mech1Points, mech1Crease, mech1Glues, false, 1);

    
    // Creating mechanism 2, a second generation symmetric v-fold
    //bll  bl/br  brr    (bl = bottomLeft, b = bottom, br = bottomRight)
    //   g___s___g      1___2___3
    //    \  |  /        \  |  /
    //     \ | /          \ | /  
    //      \|/            \|/
    //       g              0   
    //       o            (o = origin)           
    //       ^ crease line

    //float d = std::sqrt(2.0)/2.0;

    std::vector<Eigen::Vector3f> mech2Points = {
        Eigen::Vector3f{5.0f, -2.0f, 0.0f},  //0
        Eigen::Vector3f{5.0f+d, -2.0f+d, 0.0f},  //1
        Eigen::Vector3f{5.0f, -2.0f+d, 0.0f},  //2
        Eigen::Vector3f{5.0f-d, -2.0f+d, 0.0f},  //3
    };

    std::vector<Crease> mech2Crease = {
        // Crease {i, j, type}
        Crease {0, 2, CreaseType::Mountain}
    };

    std::vector<GlueDot> mech2Glues = { // this part is tricky because it's up to the artists to make sure that lengths stay the same 
        // GlueDot {i, n, gluePosition, weights}
        GlueDot {1, 1, Eigen::Vector3f{5.0f+d, 0.0f+d, 0.0f}, Eigen::Vector3f{0.0,0.0,0.0}},
        GlueDot {0, 0, Eigen::Vector3f{0.0f, 0.0f, 0.0f}, Eigen::Vector3f{0.0,0.0,0.0}},
        //GlueDot {3, 0, Eigen::Vector3f{0.0f+d, 0.0f-d, 0.0f}, Eigen::Vector3f{0.0,0.0,0.0}},
        GlueDot {3, 0, Eigen::Vector3f{0.0f+1.0f, 0.0f, 0.0f}, Eigen::Vector3f{0.0,0.0,0.0}},
    };

    Mechanism mech2(mech2Points, mech2Crease, mech2Glues, false, 2);







    
    // Creating the pop up card!!
    //        ______________________
    //       /       .   /         /
    //      /       /|\ /         /
    //     /       / | \         /
    //    /        \_!_/        /
    //   /__________/__________/

    std::vector<Mechanism> allMechs = {baseCard, mech1, mech2};
    constexpr float PI = 3.14159265358979323846f;
    PopUpCard myPopUpCard(allMechs);
